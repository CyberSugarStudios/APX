// Paste these rules into:
// Firebase Console > Firestore Database > Rules tab

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Users can read and write their own data only
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // World invite codes: any signed-in player can read a code to join,
    // but only the GM (owner) can create or delete the code entry.
    match /worldCodes/{inviteCode} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == resource.data.gmUid;
      allow create: if request.auth != null;
    }

  }
}
