// ─────────────────────────────────────────────────────────────────────────────
// APX FIRESTORE RULES  (v2026.9.22)
// Paste into: Firebase Console > Firestore Database > Rules tab
// ─────────────────────────────────────────────────────────────────────────────
// No Firebase Storage or Blaze plan required.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Users own everything under their UID ────────────────────────────────
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // ── Map image override: any signed-in player may READ a world's map ──────
    // Players discover gmUid + worldId only after joining via an invite code,
    // so this path is not practically guessable.  The wildcard rule above
    // grants write only to the owner; this more-specific rule adds read for
    // any authenticated user (Firestore: any matching rule allows the op).
    match /users/{userId}/worlds/{worldId}/mapImage/{docId} {
      allow read: if request.auth != null;
    }

    // Other map images (dungeons, buildings, etc) — same read-access for players
    match /users/{userId}/worlds/{worldId}/otherMaps/{mapId} {
      allow read: if request.auth != null;
    }

    // NPC portraits — players read the SAME full-quality (400px) portrait the GM
    // uses on battle tokens, so both maps show identical images.
    match /users/{userId}/worlds/{worldId}/npcPortraits/{npcId} {
      allow read: if request.auth != null;
    }

    // Fog of War data — players read the GM's fog to render it on their maps
    match /users/{userId}/worlds/{worldId}/fogData/{mapId} {
      allow read: if request.auth != null;
    }

    // ── World invite codes ───────────────────────────────────────────────────
    match /worldCodes/{inviteCode} {
      allow read:   if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null
          && request.auth.uid == resource.data.gmUid;
    }

    // ── Public map image (fallback / legacy) ─────────────────────────────────
    match /worldCodes/{inviteCode}/mapImage/{docId} {
      allow read:  if request.auth != null;
      allow write: if request.auth != null
          && get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.gmUid == request.auth.uid;
    }

    // ── Player character data ────────────────────────────────────────────────
    // Players write charState so the GM party panel can read live stats.
    // Banned and kicked players cannot write — prevents auto-rejoin after kick.
    // The GM writes _gmHp back to this document for initiative HP sync.
    match /worldCodes/{inviteCode}/players/{playerId} {
      allow read: if request.auth != null;
      // GM can always write (kick, HP override, etc.)
      allow write: if request.auth != null
          && get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.gmUid == request.auth.uid;
      // Players write their own doc ONLY if not banned or kicked
      allow write: if request.auth != null
          && request.auth.uid == playerId
          && !(request.auth.uid in get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.get('banned', []))
          && !(request.auth.uid in get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.get('kicked', []));
    }

  }
}
