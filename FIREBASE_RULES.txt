// ─────────────────────────────────────────────────────────────────────────────
// APX FIRESTORE RULES  (v2026.10.28)
// Paste into: Firebase Console > Firestore Database > Rules tab
// ─────────────────────────────────────────────────────────────────────────────
// No Firebase Storage or Blaze plan required.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Helpers (the Bastion rules below use them) ───────────────────────────
    function worldOf(code) { return get(/databases/$(database)/documents/worldCodes/$(code)).data; }
    function isWorldGm(code) { return request.auth != null && worldOf(code).gmUid == request.auth.uid; }
    // Someone who joined the world (they have a player record) and hasn't been kicked or banned
    function isWorldMember(code) {
      return request.auth != null
          && exists(/databases/$(database)/documents/worldCodes/$(code)/players/$(request.auth.uid))
          && !(request.auth.uid in worldOf(code).get('banned', []))
          && !(request.auth.uid in worldOf(code).get('kicked', []));
    }

    // ── Users own everything under their UID ────────────────────────────────
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // ── Map image override: any signed-in player may READ a world's map ──────
    // Players discover gmUid + worldId only after joining via an invite code,
    // so this path is not practically guessable.  The wildcard rule above
    // grants write only to the owner; this more-specific rule adds read for
    // any authenticated user (Firestore: any matching rule allows the op).
    match /users/{userId}/worlds/{worldId}/mapImage/{docId} {
      allow read: if request.auth != null;
    }

    // Other map images (dungeons, buildings, etc) — same read-access for players
    match /users/{userId}/worlds/{worldId}/otherMaps/{mapId} {
      allow read: if request.auth != null;
    }

    // NPC portraits — players read the SAME full-quality (400px) portrait the GM
    // uses on battle tokens, so both maps show identical images.
    match /users/{userId}/worlds/{worldId}/npcPortraits/{npcId} {
      allow read: if request.auth != null;
    }

    // Battle-map images (props like carts, tower floors) — players read them too
    match /users/{userId}/worlds/{worldId}/battleImages/{imgId} {
      allow read: if request.auth != null;
    }

    // Full-resolution map tiles (large maps are saved as a preview plus tiles)
    match /users/{userId}/worlds/{worldId}/mapTiles/{tileId} {
      allow read: if request.auth != null;
    }

    // Fog of War data — players read the GM's fog to render it on their maps
    match /users/{userId}/worlds/{worldId}/fogData/{mapId} {
      allow read: if request.auth != null;
    }

    // ── World invite codes ───────────────────────────────────────────────────
    match /worldCodes/{inviteCode} {
      allow read:   if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null
          && request.auth.uid == resource.data.gmUid;
    }

    // ── Public map image (fallback / legacy) ─────────────────────────────────
    match /worldCodes/{inviteCode}/mapImage/{docId} {
      allow read:  if request.auth != null;
      allow write: if request.auth != null
          && get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.gmUid == request.auth.uid;
    }

    // ── World chat (dice tray) ───────────────────────────────────────────────
    // A message is read by everyone named in `to` ('all' = the whole world) and by the world's GM.
    // You can only send as yourself, and not once you've been kicked or banned.
    // The sender and the GM can delete a message (the GM's "Clear chat").
    match /worldCodes/{inviteCode}/chat/{msgId} {
      allow read: if request.auth != null
          && (request.auth.uid in resource.data.to
              || 'all' in resource.data.to
              || get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.gmUid == request.auth.uid);
      allow create: if request.auth != null
          && request.resource.data.from == request.auth.uid
          && request.resource.data.text is string && request.resource.data.text.size() <= 1000
          && request.resource.data.to is list && request.resource.data.to.size() <= 40
          && !(request.auth.uid in get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.get('banned', []))
          && !(request.auth.uid in get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.get('kicked', []));
      allow delete: if request.auth != null
          && (resource.data.from == request.auth.uid
              || get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.gmUid == request.auth.uid);
    }

    // ── Player character data ────────────────────────────────────────────────
    // Players write charState so the GM party panel can read live stats.
    // Banned and kicked players cannot write — prevents auto-rejoin after kick.
    // The GM writes _gmHp back to this document for initiative HP sync.
    match /worldCodes/{inviteCode}/players/{playerId} {
      allow read: if request.auth != null;
      // GM can always write (kick, HP override, etc.)
      allow write: if request.auth != null
          && get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.gmUid == request.auth.uid;
      // Players write their own doc ONLY if not banned or kicked
      allow write: if request.auth != null
          && request.auth.uid == playerId
          && !(request.auth.uid in get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.get('banned', []))
          && !(request.auth.uid in get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.get('kicked', []));
      // A player may always remove their own record (leaving a world, deleting their account)
      allow delete: if request.auth != null && request.auth.uid == playerId;
    }

    // ── Bastion (Rulebook Chapter 13) ────────────────────────────────────────
    // worldCodes/{code}/bastion/main: the party's shared Bastion. Everyone signed in can read it (like
    // the world itself); only the world's GM writes it: the GM's app applies what the players ask for.
    match /worldCodes/{inviteCode}/bastion/{bastionId} {
      allow read: if request.auth != null;
      allow write: if isWorldGm(inviteCode);
    }
    // Suggestions: buildings a player draws on the Bastion map. The author writes their own (while the
    // GM can change or remove any); everyone in the world can vote, changing only their own vote.
    match /worldCodes/{inviteCode}/bastion/{bastionId}/sugs/{sugId} {
      allow read: if request.auth != null;
      allow create: if (isWorldMember(inviteCode) || isWorldGm(inviteCode)) && request.resource.data.by == request.auth.uid;
      allow update: if isWorldGm(inviteCode)
          || (isWorldMember(inviteCode) && resource.data.by == request.auth.uid && request.resource.data.by == request.auth.uid)
          || (isWorldMember(inviteCode)
              && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['votes'])
              && request.resource.data.get('votes', {}).diff(resource.data.get('votes', {})).affectedKeys().hasOnly([request.auth.uid]));
      allow delete: if isWorldGm(inviteCode) || (request.auth != null && resource.data.by == request.auth.uid);
    }
    // Requests to the GM's app (a deposit, assigning a Recruit, a Bastion check's result): sent as
    // yourself; the GM's app answers them and tidies them away.
    match /worldCodes/{inviteCode}/bastion/{bastionId}/reqs/{reqId} {
      allow read: if request.auth != null;
      allow create: if (isWorldMember(inviteCode) || isWorldGm(inviteCode)) && request.resource.data.by == request.auth.uid;
      allow update, delete: if isWorldGm(inviteCode) || (request.auth != null && resource.data.by == request.auth.uid);
    }
    // Pictures for the Bastion map (a level's background, a Plot Type's tile), under 1 MB each
    match /worldCodes/{inviteCode}/bastion/{bastionId}/imgs/{imgId} {
      allow read: if request.auth != null;
      allow create: if (isWorldMember(inviteCode) || isWorldGm(inviteCode)) && request.resource.data.by == request.auth.uid
          && request.resource.data.src is string && request.resource.data.src.size() < 1040000;
      allow delete: if isWorldGm(inviteCode) || (request.auth != null && resource.data.by == request.auth.uid);
    }
    // Choices the managers and GM make together (how Structural Damage is absorbed)
    match /worldCodes/{inviteCode}/bastion/{bastionId}/live/{docId} {
      allow read: if request.auth != null;
      allow write: if isWorldMember(inviteCode) || isWorldGm(inviteCode);
    }

    // ── Per-player Fog of War ────────────────────────────────────────────────
    // Each player's own reveals on a battle map (players/{uid}/fog/{mapId}). Everyone in the world can read
    // them (the GM views a player's fog); the player writes their own, and the GM writes anyone's.
    match /worldCodes/{inviteCode}/players/{playerId}/fog/{mapId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
          && (request.auth.uid == playerId
              || get(/databases/$(database)/documents/worldCodes/$(inviteCode)).data.gmUid == request.auth.uid);
    }

  }
}
